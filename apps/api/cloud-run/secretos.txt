# =============================================================================
# Secretos de la API en Cloud Run, desde Secret Manager · D-154 (ADR-040)
#
#   VARIABLE=nombre-del-secreto
#
# Sólo NOMBRES: el valor lo carga Argenis en Secret Manager por la entrada
# estándar (DESPLIEGUE_CLOUD_RUN.md §6). El flujo de despliegue construye con
# esta lista `--set-secrets=VARIABLE=nombre:latest,…`, que REEMPLAZA los
# secretos de la revisión: lo que no está aquí no llega a la API.
#
# Los seis primeros son los que la API exige para arrancar; seis es también lo
# que Secret Manager da gratis (6 versiones activas). El séptimo lo exige la
# consola en Netlify (D-160).
# =============================================================================
SUPABASE_SECRET_KEY=ncr-supabase-secret-key
# El pooler en modo SESIÓN (:5432). La API sólo usa DATABASE_URL para pg-boss,
# que necesita sesión (cerrojos y LISTEN); la directa sólo resuelve por IPv6.
DATABASE_URL=ncr-database-url-sesion
# El pooler en modo TRANSACCIÓN (:6543): el pool de la aplicación.
DATABASE_POOLER_URL=ncr-database-url-transaccion
INGESTA_FIRMA_SECRETO=ncr-ingesta-firma-secreto
BIOMETRIA_LLAVE=ncr-biometria-llave
EQUIPOS_LLAVE=ncr-equipos-llave
# D-160 (PR 2) · la consola en Netlify firma la IP del navegador con este
# secreto, que comparte con ella (CONSOLA_IP_FIRMA_SECRETO). Sin él, la API no
# cree la IP que declara la consola y la lista blanca de porteros niega a todos.
# Es la SÉPTIMA versión activa: la primera fuera de la capa gratuita (USD 0,06/mes).
API_IP_FIRMA_SECRETO=ncr-api-ip-firma-secreto
